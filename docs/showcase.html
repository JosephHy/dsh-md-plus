GDScript
extends CharacterBody2D

@export var speed: float = 240.0
@export var jump_velocity: float = -420.0

var gravity: float = ProjectSettings.get_setting("physics/2d/default_gravity")

func _physics_process(delta: float) -> void:
	if not is_on_floor():
		velocity.y += gravity * delta

	if Input.is_action_just_pressed("jump") and is_on_floor():
		velocity.y = jump_velocity

	var direction := Input.get_axis("move_left", "move_right")
	velocity.x = direction * speed
	move_and_slide()

03

Mermaid 图形

围栏内容直接在卡片里渲染成 SVG，支持图形 / 代码切换、缩放、拖拽、四套配色、全屏查看和 SVG / PNG 导出。

头部语言名被宿主占位成「代码块」也没关系，插件会从正文的 flowchart / sequenceDiagram 类型声明识别。

代码块
flowchart TD
A[用户提问] --> B[DSH 组装上下文]
B --> C[模型流式输出]
C --> D[解析 Markdown]
D --> E{围栏语言?}
E -->|mermaid| F[插件渲染 SVG]
E -->|其它| G[代码块卡片]
F --> H[全屏 / 缩放 / 导出]
G --> I[补高亮 / 折叠 / 复制]
代码块
sequenceDiagram
participant U as 用户
participant D as DSH
participant P as dsh-md-plus
U->>D: 发出一条消息
D->>P: 渲染 Markdown DOM
P->>P: 扫描代码块 / 表格 / 图片
P-->>D: 注入卡片按钮与样式
D-->>U: 看到成品

05

图片点击放大

旧宿主没有原生预览按钮时才由插件兜底：点击打开暗色预览，Esc 或点遮罩关闭；新版 DSH 的原生预览、头像、图标、代码块和链接图片都不接管。

06

设置面板

右下角的低调齿轮，点开可调 Mermaid 配色、代码字号、图表默认视图和三类折叠阈值；设置存在 localStorage，刷新后保持。

齿轮 点右下角按钮打开设置；把代码字号改成 16px、折叠阈值改成 0，再回来看看效果。
